# dlna-rs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-binary DLNA/UPnP media server in Rust that auto-discovers on VLC (AppleTV), with an embedded Svelte web UI for managing advertised folders.

**Architecture:** Three concurrent subsystems in one tokio runtime: SSDP responder + UPnP HTTP routes (provided by `librqbit-upnp-serve`), plus our own axum routes for `/stream/{id}` (Range + DLNA headers), JSON admin API, and an embedded Svelte SPA at `/`. A shared `Arc<RwLock<Library>>` holds folders + the `usize → PathBuf` id map.

**Tech Stack:** Rust (axum, tokio, walkdir, mime_guess, rust-embed, facet-styx, librqbit-upnp-serve, clap), Svelte 5 (vite, no SvelteKit), bun for the web build.

**Project root:** `/Users/m/dev/dlna-rs`

```rust
use anyhow::{Context, Result};
use facet::Facet;
use std::path::{Path, PathBuf};

#[derive(Facet, Debug, Clone, PartialEq, Eq)]
pub struct Folder {
	pub path: String,
	pub recursive: bool,
}

#[derive(Facet, Debug, Clone, PartialEq, Eq)]
pub struct Config {
	pub port: u16,
	pub friendly_name: String,
	pub folders: Vec<Folder>,
}

impl Config {
	pub fn with_defaults() -> Self {
		let host = gethostname::gethostname().to_string_lossy().into_owned();
		Self {
			port: 9119,
			friendly_name: format!("dlna-rs ({host})"),
			folders: Vec::new(),
		}
	}

	pub fn default_path() -> PathBuf {
		let home = std::env::var_os("HOME")
			.map(PathBuf::from)
			.unwrap_or_else(|| PathBuf::from("."));
		home.join(".config/dlna-rs/config.styx")
	}

	pub fn load_or_default(path: &Path) -> Result<Self> {
		match std::fs::read_to_string(path) {
			Ok(s) => facet_styx::from_str(&s).with_context(|| format!("parse {}", path.display())),
			Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(Self::with_defaults()),
			Err(e) => Err(e).with_context(|| format!("read {}", path.display())),
		}
	}

	pub fn save_atomic(&self, path: &Path) -> Result<()> {
		if let Some(parent) = path.parent() {
			std::fs::create_dir_all(parent)?;
		}
		let body = facet_styx::to_string(self).context("serialize config")?;
		let tmp = path.with_extension("styx.tmp");
		std::fs::write(&tmp, body)?;
		std::fs::rename(&tmp, path)?;
		Ok(())
	}
}
```

```rust
#[derive(Parser, Debug)]
#[command(name = "dlna-rs", version, about = "Single-binary DLNA media server")]

	#[arg(long)]
	pub config: Option<PathBuf>,
	#[arg(long)]
	pub port: Option<u16>,
	#[arg(long)]
	pub friendly_name: Option<String>,
	#[arg(long = "folder", value_name = "PATH")]
	pub folder: Vec<PathBuf>,
	#[arg(long = "folder-flat", value_name = "PATH")]

impl Cli {
	pub fn apply(self, cfg: &mut crate::config::Config) {
		if let Some(p) = self.port { cfg.port = p; }
		if let Some(n) = self.friendly_name { cfg.friendly_name = n; }
		for p in self.folder {
			cfg.folders.push(crate::config::Folder { path: p.to_string_lossy().into_owned(), recursive: true });
		}
		for p in self.folder_flat {
			cfg.folders.push(crate::config::Folder { path: p.to_string_lossy().into_owned(), recursive: false });
		}
	}
}
```

```rust
use anyhow::{Context, Result};
use std::collections::HashMap;
use std::path::{Path, PathBuf};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Entry {
	File(PathBuf),
	Dir(PathBuf),
}

impl Entry {
	pub fn path(&self) -> &Path {
		match self {
			Entry::File(p) | Entry::Dir(p) => p,
		}
	}
	pub fn is_dir(&self) -> bool {
		matches!(self, Entry::Dir(_))
	}
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Folder {
	pub path: PathBuf,
	pub recursive: bool,
	pub file_count: usize,
}

#[derive(Debug, Default)]
pub struct IdMap {
	forward: HashMap<usize, Entry>,
	reverse: HashMap<PathBuf, usize>,
	next: usize,
}

impl IdMap {
	pub fn intern(&mut self, entry: Entry) -> usize {
		if let Some(&id) = self.reverse.get(entry.path()) {
			return id;
		}
		self.next += 1;
		let id = self.next;
		self.reverse.insert(entry.path().to_path_buf(), id);
		self.forward.insert(id, entry);
		id
	}

	pub fn get(&self, id: usize) -> Option<&Entry> {
		self.forward.get(&id)
	}
}

#[derive(Debug, Default)]
pub struct Library {
	folders: Vec<Folder>,
	ids: IdMap,
}

impl Library {
	pub fn new() -> Self { Self::default() }

	pub fn folders(&self) -> &[Folder] { &self.folders }
	pub fn ids(&self) -> &IdMap { &self.ids }

	pub fn intern(&mut self, entry: Entry) -> usize {
		self.ids.intern(entry)
	}

	pub fn add_folder(&mut self, path: PathBuf, recursive: bool) -> Result<Folder> {
		let canonical = path.canonicalize().with_context(|| format!("canonicalize {}", path.display()))?;
		if self.folders.iter().any(|f| f.path == canonical) {
			anyhow::bail!("folder already added: {}", canonical.display());
		}
		let file_count = Self::count_files(&canonical, recursive);
		let folder = Folder { path: canonical, recursive, file_count };
		self.folders.push(folder.clone());
		Ok(folder)
	}

	pub fn remove_folder(&mut self, path: &Path) -> bool {
		let target = path.canonicalize().unwrap_or_else(|_| path.to_path_buf());
		let len = self.folders.len();
		self.folders.retain(|f| f.path != target);
		self.folders.len() != len
	}

	pub fn refresh_count(&mut self, path: &Path) -> Option<usize> {
		let target = path.canonicalize().ok()?;
		let f = self.folders.iter_mut().find(|f| f.path == target)?;
		f.file_count = Self::count_files(&f.path, f.recursive);
		Some(f.file_count)
	}

	fn count_files(root: &Path, recursive: bool) -> usize {
		let max_depth = if recursive { usize::MAX } else { 1 };
		walkdir::WalkDir::new(root)
			.max_depth(max_depth)
			.follow_links(false)
			.into_iter()
			.filter_map(|e| e.ok())
			.filter(|e| e.file_type().is_file())
			.count()
	}
}
```

```rust
impl Library {
	pub fn from_config(folders: &[crate::config::Folder]) -> Self {
		let mut lib = Self::new();
		for f in folders {
			let path = PathBuf::from(&f.path);
			if let Err(e) = lib.add_folder(path.clone(), f.recursive) {
				tracing::warn!(?path, "skipping folder: {e}");
			}
		}
		lib
	}

	pub fn to_config_folders(&self) -> Vec<crate::config::Folder> {
		self.folders.iter()
			.map(|f| crate::config::Folder {
				path: f.path.to_string_lossy().into_owned(),
				recursive: f.recursive,
			})
			.collect()
	}
}
```

```rust
pub type LibState = Arc<RwLock<Library>>;

#[derive(Serialize)]
pub struct FolderView {
	pub path: String,
	pub recursive: bool,
	pub file_count: usize,
}

#[derive(Deserialize)]
pub struct AddFolderReq {
	pub path: String,
	pub recursive: bool,
}

#[derive(Deserialize)]
pub struct RemoveFolderReq {
	pub path: String,
}

#[derive(Deserialize)]
pub struct BrowseQuery {
	pub path: Option<String>,
}

#[derive(Serialize)]
pub struct BrowseEntry {
	pub name: String,
	pub is_dir: bool,
}

#[derive(Serialize)]
pub struct BrowseResp {
	pub path: String,
	pub parent: Option<String>,
	pub entries: Vec<BrowseEntry>,
}

pub fn router(lib: LibState) -> Router {
	Router::new()
		.route("/api/folders", get(list_folders).post(add_folder).delete(remove_folder))
		.route("/api/folders/refresh", post(refresh_folder))
		.route("/api/browse", get(browse))
		.route("/api/status", get(api_status))
		.route("/status", get(status_text))
		.with_state(lib)
}

async fn list_folders(State(lib): State<LibState>) -> Json<Vec<FolderView>> {
	Json(snapshot(&*lib.read().await))
}

async fn add_folder(State(lib): State<LibState>, Json(req): Json<AddFolderReq>) -> Response {
	let mut g = lib.write().await;
	match g.add_folder(PathBuf::from(&req.path), req.recursive) {
		Ok(f) => (StatusCode::CREATED, Json(FolderView {
			path: f.path.to_string_lossy().into_owned(),
			recursive: f.recursive,
			file_count: f.file_count,
		})).into_response(),
		Err(e) => (StatusCode::BAD_REQUEST, e.to_string()).into_response(),
	}
}

async fn remove_folder(State(lib): State<LibState>, Json(req): Json<RemoveFolderReq>) -> StatusCode {
	if lib.write().await.remove_folder(&PathBuf::from(&req.path)) {
		StatusCode::NO_CONTENT
	} else {
		StatusCode::NOT_FOUND
	}
}

async fn refresh_folder(State(lib): State<LibState>, Json(req): Json<RemoveFolderReq>) -> Response {
	let mut g = lib.write().await;
	match g.refresh_count(&PathBuf::from(&req.path)) {
		Some(n) => Json(serde_json::json!({ "file_count": n })).into_response(),
		None => (StatusCode::NOT_FOUND, "folder not found").into_response(),
	}
}

async fn browse(Query(q): Query<BrowseQuery>) -> Response {
	let path = q.path
		.map(PathBuf::from)
		.unwrap_or_else(|| std::env::var_os("HOME").map(PathBuf::from).unwrap_or_else(|| PathBuf::from("/")));
	let path = match path.canonicalize() {
		Ok(p) => p,
		Err(e) => return (StatusCode::BAD_REQUEST, format!("canonicalize: {e}")).into_response(),
	};
	let mut entries: Vec<BrowseEntry> = match std::fs::read_dir(&path) {
		Ok(rd) => rd.filter_map(|e| e.ok())
			.filter_map(|e| {
				let ft = e.file_type().ok()?;
				let name = e.file_name().to_string_lossy().into_owned();
				if name.starts_with('.') { return None; }
				Some(BrowseEntry { name, is_dir: ft.is_dir() })
			})
			.collect(),
		Err(e) => return (StatusCode::BAD_REQUEST, format!("read_dir: {e}")).into_response(),
	};
	entries.sort_by(|a, b| b.is_dir.cmp(&a.is_dir).then_with(|| a.name.cmp(&b.name)));
	let parent = path.parent().map(|p| p.to_string_lossy().into_owned());
	Json(BrowseResp { path: path.to_string_lossy().into_owned(), parent, entries }).into_response()
}

async fn api_status(State(lib): State<LibState>) -> Json<serde_json::Value> {
	let g = lib.read().await;
	let folders = snapshot(&*g);
	Json(serde_json::json!({ "folders": folders }))
}

async fn status_text(State(lib): State<LibState>) -> String {
	let g = lib.read().await;
	let folders = snapshot(&*g);
	let mut s = String::new();
	for f in folders {
		s.push_str(&format!("{}  {} files\n", f.path, f.file_count));
	}
	if s.is_empty() { s.push_str("(no folders configured)\n"); }
	s
}

fn snapshot(lib: &Library) -> Vec<FolderView> {
	lib.folders().iter().map(|f| FolderView {
		path: f.path.to_string_lossy().into_owned(),
		recursive: f.recursive,
		file_count: f.file_count,
	}).collect()
}
```

```rust
use crate::library::{Entry, Library};
use librqbit_upnp_serve::services::content_directory::{
	browse::response::{Container, Item, ItemOrContainer},
	ContentDirectoryBrowseProvider,
};
use std::path::PathBuf;
use std::sync::Arc;
use tokio::sync::RwLock;

pub struct LibraryProvider(pub Arc<RwLock<Library>>);

impl ContentDirectoryBrowseProvider for LibraryProvider {
	fn browse_direct_children(&self, parent_id: usize, http_hostname: &str) -> Vec<ItemOrContainer> {
		let mut lib = self.0.blocking_write();
		if parent_id == 0 {
			let mut out = Vec::new();
			for (i, f) in lib.folders().iter().enumerate() {
				let folder_path = f.path.clone();
				let id = lib.intern(Entry::Dir(folder_path.clone()));
				out.push(ItemOrContainer::Container(Container {
					id,
					parent_id: Some(0),
					children_count: None,
					title: folder_path.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_else(|| folder_path.to_string_lossy().into_owned()),
				}));
				let _ = i;
			}
			return out;
		}
		let entry = match lib.ids().get(parent_id) {
			Some(e) => e.clone(),
			None => return Vec::new(),
		};
		let dir = match entry {
			Entry::Dir(p) => p,
			Entry::File(_) => return Vec::new(),
		};
		let read = match std::fs::read_dir(&dir) {
			Ok(r) => r,
			Err(_) => return Vec::new(),
		};
		let mut out = Vec::new();
		for de in read.flatten() {
			let path = de.path();
			let name = de.file_name().to_string_lossy().into_owned();
			if name.starts_with('.') { continue; }
			let ft = match de.file_type() { Ok(f) => f, Err(_) => continue };
			if ft.is_dir() {
				let id = lib.intern(Entry::Dir(path.clone()));
				out.push(ItemOrContainer::Container(Container {
					id,
					parent_id: Some(parent_id),
					children_count: None,
					title: name,
				}));
			} else if ft.is_file() {
				let meta = match de.metadata() { Ok(m) => m, Err(_) => continue };
				let id = lib.intern(Entry::File(path.clone()));
				let mime = mime_guess::from_path(&path).first();
				let url = format!("http://{hostname}/stream/{id}", hostname = http_hostname, id = id);
				out.push(ItemOrContainer::Item(Item {
					id,
					parent_id,
					title: name,
					mime_type: mime,
					url,
					size: meta.len(),
				}));
			}
		}
		out
	}

	fn browse_metadata(&self, object_id: usize, http_hostname: &str) -> Vec<ItemOrContainer> {
		if object_id == 0 {
			return vec![ItemOrContainer::Container(Container {
				id: 0, parent_id: None, children_count: None,
				title: "dlna-rs".to_string(),
			})];
		}
		let lib = self.0.blocking_read();
		let entry = match lib.ids().get(object_id) {
			Some(e) => e.clone(),
			None => return Vec::new(),
		};
		match entry {
			Entry::Dir(p) => vec![ItemOrContainer::Container(Container {
				id: object_id,
				parent_id: Some(0),
				children_count: None,
				title: p.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default(),
			})],
			Entry::File(p) => {
				let size = std::fs::metadata(&p).map(|m| m.len()).unwrap_or(0);
				let mime = mime_guess::from_path(&p).first();
				let url = format!("http://{http_hostname}/stream/{object_id}");
				vec![ItemOrContainer::Item(Item {
					id: object_id,
					parent_id: 0,
					title: p.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default(),
					mime_type: mime,
					url,
					size,
				})]
			}
		}
	}
}
```

```rust
use crate::api::AppState;
use crate::library::Entry;
use axum::{
	extract::{Path, State},
	http::{header, HeaderMap, HeaderValue, StatusCode},
	response::{IntoResponse, Response},
	routing::get,
	Router,
};
use std::path::PathBuf;
use tokio::io::{AsyncReadExt, AsyncSeekExt};
use tokio_util::io::ReaderStream;

const CHUNK: usize = 64 * 1024;

pub fn router(state: AppState) -> Router {
	Router::new().route("/stream/{id}", get(handle)).with_state(state)
}

async fn handle(
	State(state): State<AppState>,
	Path(id): Path<usize>,
	headers: HeaderMap,
) -> Response {
	let path: PathBuf = {
		let lib = state.lib.read().await;
		match lib.ids().get(id) {
			Some(Entry::File(p)) => p.clone(),
			_ => return (StatusCode::NOT_FOUND, "unknown id").into_response(),
		}
	};
	let mut file = match tokio::fs::File::open(&path).await {
		Ok(f) => f,
		Err(_) => return (StatusCode::NOT_FOUND, "file gone").into_response(),
	};
	let total = match file.metadata().await {
		Ok(m) => m.len(),
		Err(_) => return (StatusCode::INTERNAL_SERVER_ERROR, "stat failed").into_response(),
	};
	let mime = mime_guess::from_path(&path).first_or_octet_stream();

	let range_hdr = headers.get(header::RANGE).and_then(|h| h.to_str().ok());
	let parsed = range_hdr.and_then(|h| http_range_header::parse_range_header(h).ok());

	let (start, end_inclusive, status) = if let Some(parsed) = parsed {
		match parsed.validate(total) {
			Ok(ranges) => {
				let r = ranges.first().cloned();
				match r {
					Some(r) => (*r.start(), *r.end(), StatusCode::PARTIAL_CONTENT),
					None => return (StatusCode::RANGE_NOT_SATISFIABLE, "no ranges").into_response(),
				}
			}
			Err(_) => return (StatusCode::RANGE_NOT_SATISFIABLE, "invalid range").into_response(),
		}
	} else {
		(0u64, total.saturating_sub(1), StatusCode::OK)
	};

	if let Err(_) = file.seek(std::io::SeekFrom::Start(start)).await {
		return (StatusCode::INTERNAL_SERVER_ERROR, "seek failed").into_response();
	}
	let to_take = end_inclusive.saturating_sub(start) + 1;
	let limited = file.take(to_take);
	let stream = ReaderStream::with_capacity(limited, CHUNK);
	let body = axum::body::Body::from_stream(stream);

	let mut resp = Response::builder()
		.status(status)
		.header(header::ACCEPT_RANGES, "bytes")
		.header(header::CONTENT_TYPE, mime.essence_str())
		.header(header::CONTENT_LENGTH, to_take)
		.header("transferMode.dlna.org", "Streaming")
		.header("contentFeatures.dlna.org", "DLNA.ORG_OP=01;DLNA.ORG_CI=0");
	if status == StatusCode::PARTIAL_CONTENT {
		resp = resp.header(
			header::CONTENT_RANGE,
			HeaderValue::from_str(&format!("bytes {start}-{end_inclusive}/{total}")).unwrap(),
		);
	}
	resp.body(body).unwrap()
}
```

```rust
use clap::Parser;
use dlna_rs::{
	api::{self, AppState},
	browse::LibraryProvider,
	cli::Cli,
	config::Config,
	library::Library,
	stream, web,
};
use librqbit_upnp_serve::{UpnpServer, UpnpServerOptions};
use std::sync::Arc;
use tokio::sync::RwLock;

#[tokio::main]
async fn main() -> anyhow::Result<()> {
	tracing_subscriber::fmt()
		.with_env_filter(tracing_subscriber::EnvFilter::try_from_default_env()
			.unwrap_or_else(|_| "dlna_rs=info,librqbit_upnp_serve=info".into()))
		.init();

	let cli = Cli::parse();
	let cfg_path = cli.config.clone().unwrap_or_else(Config::default_path);
	let mut cfg = Config::load_or_default(&cfg_path)?;
	cli.apply(&mut cfg);

	let lib = Arc::new(RwLock::new(Library::from_config(&cfg.folders)));
	let state = AppState { lib: lib.clone(), cfg_path: Arc::new(cfg_path) };

	let cancel = tokio_util::sync::CancellationToken::new();
	let mut upnp = UpnpServer::new(UpnpServerOptions {
		friendly_name: cfg.friendly_name.clone(),
		http_listen_port: cfg.port,
		http_prefix: String::new(),
		browse_provider: Box::new(LibraryProvider(lib.clone())),
		cancellation_token: cancel.clone(),
	}).await?;
	let upnp_router = upnp.take_router()?;

	let app = upnp_router
		.merge(api::router(state.clone()))
		.merge(stream::router(state.clone()))
		.merge(web::router());

	let addr = format!("0.0.0.0:{}", cfg.port);
	tracing::info!(%addr, friendly = %cfg.friendly_name, "http listening");
	let listener = tokio::net::TcpListener::bind(&addr).await?;

	let ssdp_handle = tokio::spawn(async move {
		if let Err(e) = upnp.run_ssdp_forever().await {
			tracing::warn!("ssdp loop ended: {e}");
		}
	});

	let server = tokio::spawn(async move {
		axum::serve(listener, app).await
	});

	tokio::select! {
		_ = tokio::signal::ctrl_c() => {
			tracing::info!("ctrl-c, shutting down");
			cancel.cancel();
		}
		r = server => { r??; }
	}
	let _ = ssdp_handle.await;
	Ok(())
}
```

```ts
export type Folder = { path: string; recursive: boolean; file_count: number };
export type BrowseEntry = { name: string; is_dir: boolean };
export type BrowseResp = { path: string; parent: string | null; entries: BrowseEntry[] };

export async function listFolders(): Promise<Folder[]> {
	const r = await fetch('/api/folders');
	if (!r.ok) throw new Error(await r.text());
	return r.json();
}

export async function addFolder(path: string, recursive: boolean): Promise<Folder> {
	const r = await fetch('/api/folders', {
		method: 'POST',
		headers: { 'content-type': 'application/json' },
		body: JSON.stringify({ path, recursive })
	});
	if (!r.ok) throw new Error(await r.text());
	return r.json();
}

export async function removeFolder(path: string): Promise<void> {
	const r = await fetch('/api/folders', {
		method: 'DELETE',
		headers: { 'content-type': 'application/json' },
		body: JSON.stringify({ path })
	});
	if (!r.ok) throw new Error(await r.text());
}

export async function refreshFolder(path: string): Promise<{ file_count: number }> {
	const r = await fetch('/api/folders/refresh', {
		method: 'POST',
		headers: { 'content-type': 'application/json' },
		body: JSON.stringify({ path })
	});
	if (!r.ok) throw new Error(await r.text());
	return r.json();
}

export async function browse(path?: string): Promise<BrowseResp> {
	const u = path ? `/api/browse?path=${encodeURIComponent(path)}` : '/api/browse';
	const r = await fetch(u);
	if (!r.ok) throw new Error(await r.text());
	return r.json();
}
```

```rust
use std::path::PathBuf;
use std::process::Command;

fn main() {
	let manifest = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
	let web = manifest.join("web");
	if !web.exists() {
		println!("cargo:warning=web/ missing; skipping bundle build");
		return;
	}
	println!("cargo:rerun-if-changed=web/package.json");
	println!("cargo:rerun-if-changed=web/vite.config.js");
	println!("cargo:rerun-if-changed=web/index.html");
	println!("cargo:rerun-if-changed=web/src");

	if !web.join("node_modules").exists() {
		let s = Command::new("bun").arg("install").current_dir(&web).status();
		match s {
			Ok(s) if s.success() => {}
			_ => { println!("cargo:warning=bun install failed; web UI may be missing"); return; }
		}
	}
	let s = Command::new("bun").args(["run", "build"]).current_dir(&web).status();
	match s {
		Ok(s) if s.success() => {}
		_ => println!("cargo:warning=bun run build failed; web UI may be missing"),
	}
}
```
